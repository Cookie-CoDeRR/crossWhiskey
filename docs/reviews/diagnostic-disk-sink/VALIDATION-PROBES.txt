================================================================================
Source: .scratch/probes/probe_ancestor_stat.swift

```swift
import Darwin

func verifyAncestorComponent(fd: Int32, expectedEUID: uid_t) -> (isSecure: Bool, errno: Int32) {
    var sb = stat()
    if fstat(fd, &sb) != 0 {
        return (false, errno)
    }
    // Must be directory
    guard (sb.st_mode & S_IFMT) == S_IFDIR else {
        return (false, ENOTDIR)
    }
    // Must be owned by root (0) or euid
    guard sb.st_uid == 0 || sb.st_uid == expectedEUID else {
        return (false, EACCES)
    }
    // Group and other must not have write permissions
    guard (sb.st_mode & (S_IWGRP | S_IWOTH)) == 0 else {
        return (false, EACCES)
    }
    return (true, 0)
}
```
